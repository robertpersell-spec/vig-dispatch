function addToCalendar(job) { var y=job.date.split("-")[0], mo=job.date.split("-")[1], dy=job.date.split("-")[2]; var hh=job.time.split(":")[0], mm=job.time.split(":")[1]; var start = new Date(y, mo-1, dy, hh, mm); var end = new Date(start.getTime() + 3600000); var fmt = function(d) { return d.toISOString().replace(/[-:]/g,"").split(".")[0]+"Z"; }; var details = "Ref: " + job.ref + " | Driver: " + (job.driverName||"TBC") + " | Vehicle: " + (job.vehicle||"") + " | Fare: " + job.price + " | Flight: " + (job.flightNo||"N/A") + " | From: " + (job.flightFrom||"N/A") + " | Notes: " + (job.notes||"None"); var gcalUrl = "https://calendar.google.com/calendar/render?action=TEMPLATE" + "&text=" + encodeURIComponent("VIG Transfer - " + job.clientName) + "&dates=" + fmt(start) + "/" + fmt(end) + "&location=" + encodeURIComponent(job.pickup + " to " + job.dropoff) + "&details=" + encodeURIComponent(details); var existing = document.getElementById("cal-tap-banner"); if (existing) existing.remove(); var overlay = document.createElement("div"); overlay.id = "cal-tap-banner"; overlay.style.cssText = "position:fixed;inset:0;background:rgba(0,0,0,0.6);z-index:9999;display:flex;align-items:center;justify-content:center;padding:20px;"; var box = document.createElement("div"); box.style.cssText = "background:#ffffff;border-radius:16px;padding:28px 24px;max-width:320px;width:100%;text-align:center;"; var title = document.createElement("div"); title.style.cssText = "font-size:16px;font-weight:700;color:#1a3a6b;margin-bottom:8px;"; title.textContent = "Add to Calendar"; var sub = document.createElement("div"); sub.style.cssText = "font-size:13px;color:#555;margin-bottom:20px;"; sub.textContent = "Tap a button below to add this booking to your calendar"; var gcalBtn = document.createElement("a"); gcalBtn.href = gcalUrl; gcalBtn.target = "_blank"; gcalBtn.style.cssText = "display:block;background:#1a3a6b;color:#ffffff;font-weight:700;font-size:14px;padding:13px 20px;border-radius:10px;text-decoration:none;margin-bottom:10px;font-family:Arial,sans-serif;"; gcalBtn.textContent = "Add to Google Calendar"; var copyBtn = document.createElement("button"); copyBtn.style.cssText = "display:block;width:100%;background:#f0f4ff;border:1px solid #c0cce8;color:#1a3a6b;font-weight:700;font-size:14px;padding:13px 20px;border-radius:10px;cursor:pointer;margin-bottom:14px;font-family:Arial,sans-serif;"; copyBtn.textContent = "Copy Event Details"; copyBtn.onclick = function() { var text = "VIG Transfer - " + job.clientName + " Date: " + job.date + " at " + job.time + " Pickup: " + job.pickup + " Drop-off: " + job.dropoff + " Driver: " + (job.driverName||"TBC") + " Vehicle: " + (job.vehicle||"") + " Fare: " + job.price + " Flight: " + (job.flightNo||"N/A") + " From: " + (job.flightFrom||"N/A") + " Notes: " + (job.notes||"None"); navigator.clipboard.writeText(text).then(function() { copyBtn.textContent = "Copied!"; setTimeout(function() { copyBtn.textContent = "Copy Event Details"; }, 2000); }); }; var cancel = document.createElement("button"); cancel.style.cssText = "background:none;border:none;color:#888;font-size:13px;cursor:pointer;padding:8px;font-family:Arial,sans-serif;"; cancel.textContent = "Cancel"; cancel.onclick = function() { overlay.remove(); }; overlay.onclick = function(e) { if (e.target === overlay) overlay.remove(); }; box.appendChild(title); box.appendChild(sub); box.appendChild(gcalBtn); box.appendChild(copyBtn); box.appendChild(cancel); overlay.appendChild(box); document.body.appendChild(overlay); }
